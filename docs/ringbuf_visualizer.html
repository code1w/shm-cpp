TryWrite 真实实现（ringbuf.hpp，简化注释）

if (len == 0 || len > max_write_size) return -1;  // 空写 / 超上限拒绝
w = write_pos.load(relaxed);      // 写者独占 write_pos → relaxed
r = read_pos.load(acquire);       // 配对读者的 release
if ((w + len) - r > Capacity) return -1; // 空间检查（无符号减法）
phys = w & (Capacity - 1);         // 单调偏移 → 物理偏移
tail = Capacity - phys;
if (len <= tail) memcpy(base+phys, src, len);
else { memcpy(base+phys, src, tail);        // 段 1：环尾
       memcpy(base, src+tail, len-tail); }  // 段 2：环头
write_pos.store(w + len, release);  // 发布！此前数据对读者不可见
